```rust
// compute the nth fibonacci number
fn fibonacci(n: u32) -> u64 {
    match n {
        0 => 0,
        1 => 1,
        _ => fibonacci(n - 1) + fibonacci(n - 2),
    }
}

fn main() {
    let label = "fibonacci(10)";
    println!("{label} = {}", fibonacci(10));
}
```

```python
# simple palette lookup
from dataclasses import dataclass

@dataclass
class Swatch:
    name: str
    shade: str
    hex: str

def describe(swatch: Swatch) -> str:
    return f"{swatch.name}.{swatch.shade} -> {swatch.hex}"

if __name__ == "__main__":
    print(describe(Swatch("yellow", "base", "#D1C3A9")))
```
